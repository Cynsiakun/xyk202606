CREATE TABLE `asset_fingerprint_rule` (
  `id` bigint(20) NOT NULL AUTO_INCREMENT,
  `rule_code` varchar(64) NOT NULL COMMENT '规则编码',
  `name` varchar(255) NOT NULL COMMENT '规则名称',
  `category` varchar(32) NOT NULL COMMENT '资产类别：middleware/database/bigdata/webapp/webframework/website',
  `sub_category` varchar(64) DEFAULT NULL COMMENT '子类别：如 tomcat/nginx/mysql/redis',
  `protocol` varchar(16) DEFAULT 'tcp' COMMENT '协议',
  `port` int(11) DEFAULT NULL COMMENT '常见端口（NULL表示不限制）',
  `banner_regex` varchar(500) DEFAULT NULL COMMENT 'Banner匹配正则',
  `vendor` varchar(100) DEFAULT NULL COMMENT '厂商',
  `product` varchar(255) NOT NULL COMMENT '产品名',
  `version_expr` varchar(255) DEFAULT NULL COMMENT '版本表达式',
  `confidence` int(11) DEFAULT 80 COMMENT '置信度 1-100',
  `description` varchar(2000) DEFAULT NULL,
  `enabled` tinyint(4) DEFAULT 1,
  `priority` int(11) DEFAULT 100 COMMENT '优先级（数字越小越优先）',
  `created_at` datetime DEFAULT current_timestamp(),
  `updated_at` datetime DEFAULT current_timestamp() ON UPDATE current_timestamp(),
  PRIMARY KEY (`id`),
  UNIQUE KEY `rule_code` (`rule_code`),
  KEY `idx_fingerprint_enabled` (`enabled`),
  KEY `idx_fingerprint_category` (`category`),
  KEY `idx_fingerprint_port` (`port`)
) ENGINE=InnoDB AUTO_INCREMENT=101 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci COMMENT='资产指纹识别规则库';
